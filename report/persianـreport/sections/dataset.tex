\section{توصیف مجموعه‌داده}

مجموعه‌دادهٔ مورد استفاده در این پروژه، مجموعه‌دادهٔ بیماری قلبی از مخزن
\lr{UCI Machine Learning Repository} \cite{uci} است. این داده‌ها شامل اطلاعات بیماران
جمع‌آوری‌شده از چهار مرکز پزشکی مختلف شامل کلیولند، مجارستان، سوئیس و مرکز \lr{VA Long Beach}
می‌باشد. هر یک از این زیرمجموعه‌ها شامل ویژگی‌های بالینی مشابهی هستند که وضعیت سلامت بیماران
و وجود یا عدم وجود بیماری قلبی را توصیف می‌کنند.

در این پروژه از نسخهٔ تجمیع‌شدهٔ مجموعه‌داده استفاده شده است که شامل حدود \textbf{۱۰۰۰}
نمونه و \textbf{۱۳} ویژگی اصلی است. علاوه بر این، یک متغیر هدف دودویی وجود دارد که نشان می‌دهد
آیا بیمار به بیماری قلبی مبتلا است (مقدار \lr{1}) یا خیر (مقدار \lr{0}). ویژگی‌های موجود را
می‌توان در سه گروه اصلی طبقه‌بندی کرد:

\begin{itemize}
	\item \textbf{ویژگی‌های عددی پیوسته}، مانند:
	\begin{itemize}
		\item \lr{age}: سن بر حسب سال
		\item \lr{trestbps}: فشار خون استراحت (بر حسب میلی‌متر جیوه)
		\item \lr{chol}: میزان کلسترول سرم (بر حسب میلی‌گرم بر دسی‌لیتر)
		\item \lr{thalach}: بیشینهٔ ضربان قلب ثبت‌شده
		\item \lr{oldpeak}: میزان افت قطعهٔ \lr{ST} ناشی از ورزش
	\end{itemize}
	
	\item \textbf{ویژگی‌های دودویی یا دسته‌ای رمزگذاری‌شده}:
	\begin{itemize}
		\item \lr{sex}: جنسیت (۰ = زن، ۱ = مرد)
		\item \lr{cp}: نوع درد قفسهٔ سینه (۰ تا ۳)
		\item \lr{fbs}: قند خون ناشتا بالاتر از ۱۲۰ (۱ = بله، ۰ = خیر)
		\item \lr{restecg}: نتیجهٔ نوار قلب استراحتی
		\item \lr{exang}: آنژین ناشی از ورزش (۱ = بله، ۰ = خیر)
		\item \lr{slope}: شیب قطعهٔ \lr{ST} در اوج فعالیت
		\item \lr{ca}: تعداد عروق اصلی (۰ تا ۳) مشاهده‌شده در فلوروسکوپی
		\item \lr{thal}: وضعیت تالاسمی (مقادیر رمزگذاری‌شده ۰ تا ۳)
	\end{itemize}
	
	\item \textbf{متغیر هدف}:
	\begin{itemize}
		\item \lr{target}: وجود بیماری قلبی (۱ = وجود بیماری، ۰ = عدم وجود بیماری)
	\end{itemize}
\end{itemize}

در پیاده‌سازی، مجموعه‌داده به صورت فایل \lr{CSV} و با استفاده از کتابخانهٔ \lr{pandas}
بارگذاری شده است. بررسی اولیهٔ داده‌ها شامل مشاهدهٔ ابعاد مجموعه‌داده، نوع ستون‌ها، و مقدار
داده‌های گمشده انجام گرفته تا اطمینان حاصل شود که داده به‌درستی ادغام شده و برای مرحلهٔ
پیش‌پردازش آماده است.
